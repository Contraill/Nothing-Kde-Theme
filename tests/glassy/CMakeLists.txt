cmake_minimum_required(VERSION 3.20)
project(GlassyChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test Svg UiTools)
find_package(KF6Config REQUIRED)
find_package(KF6ConfigWidgets REQUIRED)
find_package(KF6Svg REQUIRED)
add_executable(glassy-gallery ../glassy-gallery.cpp)
add_executable(glassy-density ../glassy-density.cpp)
target_link_libraries(glassy-density PRIVATE Qt6::Widgets)
target_link_libraries(glassy-gallery PRIVATE Qt6::Widgets Qt6::Test)
add_executable(glassy-svg ../glassy-svg.cpp)
target_link_libraries(glassy-svg PRIVATE Qt6::Widgets Qt6::Svg)
add_executable(glassy-config ../glassy-config.cpp)
target_link_libraries(glassy-config PRIVATE Qt6::Widgets Qt6::UiTools KF6::ConfigGui KF6::ConfigWidgets)
add_executable(glassy-mask ../glassy-mask.cpp)
add_executable(panel-config ../panel-config.cpp)
target_link_libraries(panel-config PRIVATE KF6::ConfigGui)
target_link_libraries(glassy-mask PRIVATE Qt6::Widgets KF6::Svg)
add_library(glassy-live-capture SHARED ../glassy-live-capture.cpp)
target_link_libraries(glassy-live-capture PRIVATE Qt6::Widgets)
