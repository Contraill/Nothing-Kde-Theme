cmake_minimum_required(VERSION 3.20)
project(NothingKDEWidgetCheck LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Quick Qml)
find_package(KF6Config REQUIRED)
find_package(Plasma REQUIRED)
find_package(PlasmaQuick REQUIRED)
find_package(KF6I18n REQUIRED)
add_executable(nothing-kde-widget-render main.cpp)
target_link_libraries(nothing-kde-widget-render PRIVATE Qt6::Widgets Qt6::Quick Qt6::Qml Plasma::Plasma Plasma::PlasmaQuick KF6::I18n)
