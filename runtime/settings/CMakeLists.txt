cmake_minimum_required(VERSION 3.20)
project(GlassySettings LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets UiTools)
find_package(KF6Config REQUIRED)
find_package(KF6ConfigWidgets REQUIRED)
find_package(KF6I18n REQUIRED)
add_executable(glassy-settings main.cpp)
target_link_libraries(glassy-settings PRIVATE Qt6::Widgets Qt6::UiTools KF6::ConfigGui KF6::ConfigWidgets KF6::I18n)
